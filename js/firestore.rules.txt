// RR Location — règles de sécurité Firestore
// À coller dans : Console Firebase → Firestore Database → Règles → Publier
//
// Tout le monde peut LIRE les disponibilités (nécessaire pour que le
// calendrier du site public fonctionne). Seul un utilisateur connecté
// (vous, via admin.html) peut ÉCRIRE dessus.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /rrlocation/disponibilites {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
